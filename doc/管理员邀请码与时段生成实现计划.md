# 管理员邀请码与时段生成 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add safe administrator invite-code binding, weekly schedule generation for the next 14 days, and appointment capacity consistency.

**Architecture:** Keep the existing Taro React frontend and WeChat cloud-function backend. Put final security and data validation in cloud functions, while the frontend only provides inputs and user feedback. Reuse existing collections and add small helper functions inside cloud functions instead of introducing new services.

**Tech Stack:** Taro 4.1.9, React 18, TypeScript 5, SCSS Modules, WeChat Cloud Functions, WeChat Cloud Database.

## Global Constraints

- Continue using WeChat cloud development cloud functions and cloud database.
- Do not rebuild the whole admin UI.
- Do not implement complex scheduling rules such as multi-segment business hours, holiday templates, or cross-day slots.
- Keep existing core collection names: `services`, `appointments`, `time_slots`.
- Admin binding must require an invite code entered in the mini program.
- Weekly fixed slots must generate concrete `time_slots` records for the next 14 days.
- Cloud functions must perform final permission and data validation.

- [ ] **Step 1: Replace open admin binding with invite-code validation**

Implement `bindAdmin` with this structure:

```js
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()

function isExpired(expiresAt) {
  if (!expiresAt) return false
  return new Date(expiresAt).getTime() < Date.now()
}

exports.main = async (event) => {
  const wxContext = cloud.getWXContext()
  const openid = wxContext.OPENID
  const inviteCode = String(event.inviteCode || '').trim()

  try {
    if (!openid) {
      return { code: -1, message: '用户身份无效', data: null }
    }
    if (!inviteCode) {
      return { code: -1, message: '请输入邀请码', data: null }
    }

    const userRes = await db.collection('users').where({ _openid: openid }).get()
    const existingUser = userRes.data[0]
    if (existingUser?.isAdmin) {
      return { code: 0, message: 'success', data: { isAdmin: true } }
    }

    const inviteRes = await db.collection('admin_invites').where({
      code: inviteCode,
      isActive: true
    }).limit(1).get()
    const invite = inviteRes.data[0]

    if (!invite) {
      await db.collection('admin_invite_logs').add({
        data: { _openid: openid, inviteId: '', result: 'invalid_code', usedAt: db.serverDate() }
      })
      return { code: -1, message: '邀请码无效', data: null }
    }
    if (isExpired(invite.expiresAt)) {
      await db.collection('admin_invite_logs').add({
        data: { _openid: openid, inviteId: invite._id, result: 'expired', usedAt: db.serverDate() }
      })
      return { code: -1, message: '邀请码已过期', data: null }
    }
    if (invite.maxUses && (invite.usedCount || 0) >= invite.maxUses) {
      await db.collection('admin_invite_logs').add({
        data: { _openid: openid, inviteId: invite._id, result: 'max_uses_reached', usedAt: db.serverDate() }
      })
      return { code: -1, message: '邀请码已达到使用次数', data: null }
    }

    if (existingUser) {
      await db.collection('users').doc(existingUser._id).update({
        data: { isAdmin: true, updateTime: db.serverDate() }
      })
    } else {
      await db.collection('users').add({
        data: {
          _openid: openid,
          nickname: '',
          avatar: '',
          phone: '',
          isAdmin: true,
          createTime: db.serverDate()
        }
      })
    }

    await db.collection('admin_invites').doc(invite._id).update({
      data: {
        usedCount: db.command.inc(1),
        updatedAt: db.serverDate()
      }
    })
    await db.collection('admin_invite_logs').add({
      data: { _openid: openid, inviteId: invite._id, result: 'success', usedAt: db.serverDate() }
    })

    return { code: 0, message: 'success', data: { isAdmin: true } }
  } catch (err) {
    console.error('[bindAdmin] error:', err)
    return { code: -1, message: err.message || '服务异常', data: null }
  }
}
```

```js
function formatDateKey(date) {
  const year = date.getFullYear()
  const month = String(date.getMonth() + 1).padStart(2, '0')
  const day = String(date.getDate()).padStart(2, '0')
  return `${year}-${month}-${day}`
}

function getNextDates(count) {
  const dates = []
  const now = new Date()
  for (let i = 0; i < count; i += 1) {
    const date = new Date(now)
    date.setDate(now.getDate() + i)
    dates.push({
      date: formatDateKey(date),
      dayOfWeek: date.getDay()
    })
  }
  return dates
}

function normalizeWeeklyConfigs(configs) {
  if (!Array.isArray(configs)) return []
  return configs
    .filter(config => Number.isInteger(config.dayOfWeek))
    .map(config => ({
      dayOfWeek: config.dayOfWeek,
      isActive: !!config.isActive,
      startTime: String(config.startTime || '').trim(),
      endTime: String(config.endTime || '').trim(),
      maxCount: Number(config.maxCount || 1)
    }))
}
```

- [ ] **Step 2: Implement `updateWeekly` branch**

Inside `exports.main`, after admin permission check and before `create`, add:

```js
    if (action === 'updateWeekly') {
      const configs = normalizeWeeklyConfigs(data.configs)
      const dates = getNextDates(14)
      let generatedCount = 0
      let updatedCount = 0
      let closedCount = 0

      for (const config of configs) {
        const matchingDates = dates.filter(item => item.dayOfWeek === config.dayOfWeek)
        for (const dateItem of matchingDates) {
          const existingRes = await db.collection('time_slots').where({
            date: dateItem.date,
            startTime: config.startTime,
            endTime: config.endTime
          }).limit(1).get()
          const existing = existingRes.data[0]

          if (config.isActive && config.startTime && config.endTime) {
            if (existing) {
              await db.collection('time_slots').doc(existing._id).update({
                data: {
                  maxCount: config.maxCount,
                  isAvailable: true,
                  source: 'weekly',
                  weeklyDayOfWeek: config.dayOfWeek,
                  updateTime: db.serverDate()
                }
              })
              updatedCount += 1
            } else {
              await db.collection('time_slots').add({
                data: {
                  date: dateItem.date,
                  startTime: config.startTime,
                  endTime: config.endTime,
                  maxCount: config.maxCount,
                  bookedCount: 0,
                  isAvailable: true,
                  source: 'weekly',
                  weeklyDayOfWeek: config.dayOfWeek,
                  createTime: db.serverDate()
                }
              })
              generatedCount += 1
            }
          }
        }

        if (!config.isActive) {
          for (const dateItem of matchingDates) {
            const slotsRes = await db.collection('time_slots').where({
              date: dateItem.date,
              source: 'weekly',
              weeklyDayOfWeek: config.dayOfWeek
            }).get()
            for (const slot of slotsRes.data) {
              if (slot.isAvailable) {
                await db.collection('time_slots').doc(slot._id).update({
                  data: { isAvailable: false, updateTime: db.serverDate() }
                })
                closedCount += 1
              }
            }
          }
        }
      }

      return { code: 0, message: 'success', data: { generatedCount, updatedCount, closedCount } }
    }
```
